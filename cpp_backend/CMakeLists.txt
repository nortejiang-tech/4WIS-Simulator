cmake_minimum_required(VERSION 3.20)
project(sim4wis_cpp VERSION 0.107.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Deterministic source bundle: pinned upstream headers/sources, no network.
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sources.tar.gz" DEPENDENCY_HASH)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sources.sha256" EXPECTED_DEPENDENCY_HASH)
string(SUBSTRING "${EXPECTED_DEPENDENCY_HASH}" 0 64 EXPECTED_DEPENDENCY_HASH)
if(NOT DEPENDENCY_HASH STREQUAL EXPECTED_DEPENDENCY_HASH)
 message(FATAL_ERROR "Third-party source bundle checksum mismatch")
endif()
set(THIRD_PARTY_DIR "${CMAKE_CURRENT_BINARY_DIR}/third_party")
file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sources.tar.gz" DESTINATION "${THIRD_PARTY_DIR}")
add_library(fourwis_core STATIC src/model.cpp src/plugin.cpp src/controller.cpp src/random.cpp src/fault.cpp src/simulation.cpp src/tracking.cpp src/steering.cpp src/path.cpp src/storage.cpp src/kpi.cpp src/agent.cpp src/live.cpp src/service.cpp src/analysis.cpp src/expression.cpp src/procedures.cpp src/criteria.cpp src/targets.cpp src/sizing.cpp src/study.cpp src/golden.cpp)
target_include_directories(fourwis_core PUBLIC include ${THIRD_PARTY_DIR}/json/single_include ../backend/native ${THIRD_PARTY_DIR}/numpy-random ${THIRD_PARTY_DIR}/eigen)
target_compile_definitions(fourwis_core PUBLIC EIGEN_DONT_VECTORIZE EIGEN_MPL2_ONLY)
if(MSVC)
 target_compile_options(fourwis_core PUBLIC /fp:strict /utf-8 /W4)
else()
 target_compile_options(fourwis_core PUBLIC -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
add_executable(sim4wis-cpp-probe src/probe.cpp)
target_link_libraries(sim4wis-cpp-probe PRIVATE fourwis_core)

file(GLOB YAML_SOURCES CONFIGURE_DEPENDS ${THIRD_PARTY_DIR}/yaml-cpp/src/*.cpp)
add_library(fourwis_yaml STATIC ${YAML_SOURCES})
target_include_directories(fourwis_yaml PUBLIC ${THIRD_PARTY_DIR}/yaml-cpp/include PRIVATE ${THIRD_PARTY_DIR}/yaml-cpp/src)
target_compile_definitions(fourwis_yaml PUBLIC YAML_CPP_STATIC_DEFINE)
target_link_libraries(fourwis_core PUBLIC fourwis_yaml)

add_library(fourwis_http STATIC ${THIRD_PARTY_DIR}/civetweb/src/civetweb.c src/http.cpp)
target_include_directories(fourwis_http PUBLIC ${THIRD_PARTY_DIR}/civetweb/include)
target_compile_definitions(fourwis_http PRIVATE NO_SSL NO_CGI NO_CACHING USE_WEBSOCKET)
target_link_libraries(fourwis_http PUBLIC fourwis_core)
if(WIN32)
 target_link_libraries(fourwis_http PUBLIC ws2_32)
else()
 find_package(Threads REQUIRED)
 target_link_libraries(fourwis_http PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()

add_executable(sim4wis-cpp src/main.cpp src/mcp.cpp)
target_link_libraries(sim4wis-cpp PRIVATE fourwis_http)

# Identify the actual compiled source independently of the checkout's HEAD.
file(GLOB_RECURSE FINGERPRINT_INPUTS CONFIGURE_DEPENDS include/*.hpp src/*.cpp data/*.json)
list(APPEND FINGERPRINT_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/../backend/native/core.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/manifest.json" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sources.sha256")
list(SORT FINGERPRINT_INPUTS)
set(FINGERPRINT_TEXT "")
foreach(INPUT_FILE IN LISTS FINGERPRINT_INPUTS)
 file(READ "${INPUT_FILE}" INPUT_TEXT)
 string(REPLACE "\r\n" "\n" INPUT_TEXT "${INPUT_TEXT}")
 string(SHA256 INPUT_HASH "${INPUT_TEXT}")
 file(RELATIVE_PATH INPUT_NAME "${CMAKE_CURRENT_SOURCE_DIR}" "${INPUT_FILE}")
 string(APPEND FINGERPRINT_TEXT "${INPUT_NAME}:${INPUT_HASH}\n")
endforeach()
string(SHA256 SOURCE_FINGERPRINT "${FINGERPRINT_TEXT}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT SOURCE_REVISION)
 set(SOURCE_REVISION "unavailable")
endif()
configure_file(build_info.hpp.in generated/build_info.hpp @ONLY)
target_include_directories(fourwis_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
